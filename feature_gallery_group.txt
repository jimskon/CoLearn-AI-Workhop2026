\section{7. Inline AI Help}{4}
\questiongroup{7.1 Four AI help modes}
\text{An inline AI help block is not a question. It sits between questions, gives students a box to ask an AI for help, and is never graded. Its mode sets the starting style, the guardrail limits what it may say, and the context lists what it can see.}
\question{7.1a What does this function return for mystery(4)? Explain how you know.}
\pythondisplay
def mystery(n):
    total = 0
    for i in range(1, n + 1):
        total += i
    return total
\endpython
\textresponse{3}
\sampleresponses{10, because it adds 1 + 2 + 3 + 4.}
\feedbackprompt{Accept 10 with a reason that mentions adding 1 through 4. If they say 6, ask whether range(1, n + 1) includes n.}
\questiontype{output_prediction}
\endquestion

\ai{explain}
\aimodel{gpt-5-mini}
\aititle{Explain: AI Coach}
\aiprompt{Not sure what the code does? Ask the coach to explain one line.}
\aiguardrail{Explain what individual lines do. Do not state the return value of mystery(4).}
\aicontext{current-question,current-code,student-response}
\aiinput{3}
\endai

\ai{critique}
\aimodel{gpt-4o-mini}
\aititle{Critique: Check Our Reasoning}
\aiprompt{Paste your explanation and ask what is missing.}
\aiguardrail{Point out one gap in the reasoning. Do not supply the corrected answer.}
\aicontext{current-question,student-response}
\aiinput{4}
\endai

\ai{testgen}
\aititle{Test Generator}
\aiprompt{Ask for test inputs that would check your prediction.}
\aiguardrail{Suggest inputs only. Do not give the expected outputs.}
\aicontext{current-question,current-code}
\aiinput{2}
\endai

\ai{generate}
\aititle{Generate: Another Example}
\aiprompt{Ask for a similar function to practice on.}
\aiguardrail{Generate a new practice function of similar difficulty. Do not solve the current question.}
\aicontext{nearby-text}
\aiinput{2}
\endai
\endquestiongroup

\section{8. Research Tags}
\questiongroup{8.1 Question types}
\text{Every question in this gallery carries an optional research tag (questiontype). Tags never change what students see or how answers are judged; they feed the Research Statistics pages. The eight values are code_writing, code_reading, output_prediction, debugging, conceptual_explanation, application_problem_solving, reflection, and other. This section has no time budget, to show that minutes are optional.}
