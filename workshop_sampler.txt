%% Section 1: Concept check
\section{Explaining the Loop}{10}
\text{A \textbf{counter-controlled loop} repeats a fixed number of times. In Python, for loops with range() are the standard tool.}

\question{In your own words, explain what a counter-controlled loop does and how Python's for loop with range() implements it. Give a concrete example with specific numbers.}
  \responsemode{answer}
  \textresponse{5}
  \sampleresponses{A counter-controlled loop runs a set number of times. Python's for loop with range(start, stop, step) handles the counting automatically. For example, range(1, 6) produces 1, 2, 3, 4, 5 — the stop value is excluded. The loop variable takes each value in turn without needing a separate counter variable.}
  \feedbackprompt{Accept if the response explains: (1) a fixed number of repetitions and (2) how range() generates the sequence. Nudge if they omit that stop is excluded or give no example. Flag if too vague.}
\endquestion

%% Section 2: Python range
\section{Python — Rewriting a Loop}{12}
\text{Below is a loop written with a literal list. Rewrite it using range() to produce identical output.}

\question{Rewrite the loop below using range() so it prints the same values (5, 10, 15, 20, 25, 30).}
  \responsemode{answer}
  \python
for value in [5, 10, 15, 20, 25, 30]:
    print(value, end=" ")

# Rewrite using range() below:
  \endpython
  \sampleresponses{for value in range(5, 31, 5):
    print(value, end=" ")}
  \feedbackprompt{Require range() generating 5,10,15,20,25,30. Stop must be ≥31. Step must be 5. Nudge if stop is 30 (misses 30).}
\endquestion

%% Section 3: Turtle
\section{Python Turtle — Filled Square}{12}

\question{Use a for loop to draw a filled square with side length 100. Fill it with any color you choose.}
  \responsemode{answer}
  \pythonturtle{400x300}
import turtle
t = turtle.Turtle()
t.speed(3)
# Your code here:
  \endpythonturtle
  \sampleresponses{t.fillcolor("blue")
t.begin_fill()
for _ in range(4):
    t.forward(100)
    t.left(90)
t.end_fill()}
  \feedbackprompt{Require: (1) for loop runs 4 times, (2) forward(100) and left(90) inside loop, (3) begin_fill() before and end_fill() after. Accept any fill color. Nudge if fill calls missing or misplaced.}
\endquestion

%% Section 4: C++
\section{C++ — Odd Numbers}{12}

\question{Write a C++ for loop that prints the odd numbers from 1 through 19, each on its own line, using cout.}
  \responsemode{answer}
  \cpp
#include <iostream>
using namespace std;

    // Write your for loop here:

  \endcpp
  \sampleresponses{for (int i = 1; i <= 19; i += 2) {

\endquestion
